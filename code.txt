  <script>
    // State management
    let classSections = ["Grade 9 - Section A", "Grade 9 - Section B", "Grade 9 - Honors"];

    function renderClasses() {
      const teacherSelect = document.getElementById("activeClassSelect");
      const studentSelect = document.getElementById("studentClassSelect");
      
      teacherSelect.innerHTML = classSections.map(c => `<option value="${c}">${c}</option>`).join('');
      studentSelect.innerHTML = classSections.map(c => `<option value="${c}">${c}</option>`).join('');
    }

    function openClassModal() { document.getElementById("classModal").classList.remove("hidden"); document.getElementById("classModal").classList.add("flex"); }
    function closeClassModal() { document.getElementById("classModal").classList.add("hidden"); document.getElementById("classModal").classList.remove("flex"); }

    function addClassSection() {
      const input = document.getElementById("newClassName");
      if(input.value.trim() !== "") {
        classSections.push(input.value.trim());
        renderClasses();
        input.value = "";
        closeClassModal();
      }
    }

    function deleteSelectedClass() {
      const select = document.getElementById("activeClassSelect");
      if(classSections.length <= 1) {
        alert("You must keep at least one active class section.");
        return;
      }
      classSections = classSections.filter(c => c !== select.value);
      renderClasses();
    }

    function switchView(view) {
      const tPortal = document.getElementById("teacherPortal");
      const sPortal = document.getElementById("studentPortal");
      const tBtn = document.getElementById("viewTeacherBtn");
      const sBtn = document.getElementById("viewStudentBtn");

      if(view === 'teacher') {
        tPortal.classList.remove("hidden");
        sPortal.classList.add("hidden");
        tBtn.className = "px-4 py-2 text-sm font-semibold rounded-lg bg-white text-indigo-700 shadow";
        sBtn.className = "px-4 py-2 text-sm font-semibold rounded-lg bg-indigo-800 text-indigo-100 hover:bg-indigo-600 transition";
      } else {
        sPortal.classList.remove("hidden");
        tPortal.classList.add("hidden");
        sBtn.className = "px-4 py-2 text-sm font-semibold rounded-lg bg-white text-indigo-700 shadow";
        tBtn.className = "px-4 py-2 text-sm font-semibold rounded-lg bg-indigo-800 text-indigo-100 hover:bg-indigo-600 transition";
      }
    }

    function generateLessonPlan() {
      const topic = document.getElementById("lpTopic").value || "Grade 9 Literary Analysis";
      const text = document.getElementById("lpText").value || "Class Curriculum Text";
      
      document.getElementById("lpTitleOut").innerText = `5E Plan: ${topic} (${text})`;
      document.getElementById("lpBodyOut").innerHTML = `
        <p><strong>Engage (7m):</strong> Hook students with imagery sample from <em>${text}</em>.</p>
        <p><strong>Explore (12m):</strong> Group annotation focused on key devices.</p>
        <p><strong>Explain (10m):</strong> Model direct analysis formula: Device + Context = Theme.</p>
        <p><strong>Elaborate (13m):</strong> Partner worksheet linking excerpts to character motives.</p>
        <p><strong>Evaluate (8m):</strong> Independent exit ticket assessment.</p>
      `;
      document.getElementById("lpResult").classList.remove("hidden");
    }

    function publishExam() {
      const title = document.getElementById("examTitle").value || "Grade 9 Quick Assessment";
      alert(`Successfully published "${title}" to all student portals!`);
    }

    function submitExam(e) {
      e.preventDefault();
      const name = document.getElementById("studentName").value;
      const list = document.getElementById("submissionsList");
      
      const newSub = document.createElement("div");
      newSub.className = "p-3 bg-slate-50 rounded-lg border border-slate-200 flex justify-between items-center";
      newSub.innerHTML = `
        <div>
          <p class="font-semibold text-slate-800">${name}</p>
          <p class="text-xs text-slate-500">Submitted just now • <span class="text-emerald-600 font-semibold">100%</span></p>
        </div>
        <span class="text-xs bg-emerald-100 text-emerald-800 px-2 py-0.5 rounded font-medium">Auto-Graded</span>
      `;
      list.prepend(newSub);
      alert("Exam submitted successfully! Your grade has been logged.");
      switchView('teacher');
    }

    // Initialize
    renderClasses();
  </script>
